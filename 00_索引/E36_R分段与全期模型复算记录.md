```r
library(foreign)
v<-read.spss(file.path(Sys.getenv("R_AUDIT_WORK"),"study1.sav"),to.data.frame=TRUE,use.value.labels=FALSE)
for(y in c("pt","ec"))for(g in c(-1,0,1)){
 a<-v[v$split.1999.2007.cubic==g,c(y,"c.data.year","num.ps")];a<-a[complete.cases(a),];fit<-lm(a[[y]]~a$c.data.year,weights=a$num.ps)
 x<-a$c.data.year;z<-a[[y]];w<-a$num.ps;mx<-weighted.mean(x,w);my<-weighted.mean(z,w);beta<-coef(fit)[2]*sqrt(sum(w*(x-mx)^2)/sum(w*(z-my)^2))
 cat(y,g,"n",nrow(a),"weight_sum",sum(w),"b",format(coef(fit)[2],digits=12),"beta",format(beta,digits=12),"\n")
}

```

### 本次实际驱动

```r
library(foreign)
v<-read.spss('tmp/e36_inputs/Study 1 data to share.sav',to.data.frame=TRUE,use.value.labels=FALSE)
years<-c('c.data.year','c.data.year.squared','c.data.year.cubed')
sets<-list(male='percent.maleMEANSUB',white='percent.whiteMEANSUB',published='published',political='right.political.view',all=c('percent.maleMEANSUB','percent.whiteMEANSUB','published','right.political.view'))
out<-list()
for(y in c('pt','ec'))for(tag in names(sets)){
 covs<-sets[[tag]];a<-v[complete.cases(v[c(y,'num.ps',years,covs)]),];stopifnot(all(a$num.ps>0))
 fit<-lm(reformulate(c(covs,years),y),data=a,weights=num.ps);base<-lm(reformulate(years,y),data=a,weights=num.ps);z<-coef(summary(fit))['c.data.year.cubed',]
 out[[length(out)+1]]<-data.frame(outcome=y,covariates=tag,n=nobs(fit),weight_sum=sum(a$num.ps),rank=fit$rank,cubic_b=z[1],cubic_se=z[2],cubic_t=z[3],nominal_p=z[4],R2=summary(fit)$r.squared,base_same_case_R2=summary(base)$r.squared,base_same_case_cubic_b=coef(base)['c.data.year.cubed'])
}
r<-do.call(rbind,out);write.csv(r,'tmp/e36_background_R_20261117.csv',row.names=FALSE);print(r,digits=10,row.names=FALSE)
```
