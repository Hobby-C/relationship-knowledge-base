复算步骤（对上述源文件；只读，不生成改写表格）：

```python
import pandas as pd
p = pd.read_excel('Participant data.xlsx', sheet_name='tempie')
s = pd.read_excel('Statement data.xlsx', sheet_name='Sheet1')
assert p.id.is_unique and s.Statement.is_unique
condition = s.set_index('Statement').Condition.to_dict()
rows = []
for _, r in p.iterrows():
    shown = {int(r[c]) for c in ['StimulusA','StimulusB','StimulusC','StimulusD']}
    observed = {i for i in range(1,17) if pd.notna(r[f'Veracity_{i}'])}
    assert observed == shown and len(observed) == 4
    for i in observed:
        assert pd.notna(r[f'Confidence_{i}'])
        rows.append((r.id, i, r[f'Veracity_{i}'], condition[i]))
d = pd.DataFrame(rows, columns=['id','statement','judgment','condition'])
assert not d.duplicated(['id','statement']).any()
d['correct'] = d.judgment.eq(d.condition)
print(pd.crosstab(d.condition, d.judgment))
print(d.correct.sum(), len(d), 100*d.correct.mean())
print(d.groupby('condition').correct.agg(['sum','count','mean']))
```
